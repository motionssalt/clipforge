# --- Transcription stack -----------------------------------------------------
# faster-whisper drives Stage A's Silero-VAD-gated Whisper transcription.
# Its VAD filter (vad_filter=True in transcribe.py) needs onnxruntime to be
# importable at runtime.
faster-whisper==1.0.3
onnxruntime>=1.17,<2

# --- Imaging -----------------------------------------------------------------
# Pillow is used by brand_scene.py / brand_scenes.py for the overlay
# composition (avatar, username, hashtags) that Stage B burns onto the
# final MP4.
Pillow>=10.3,<12

# --- Numerics ----------------------------------------------------------------
# numpy is a first-class dependency of faster-whisper and of a couple of
# our own helpers (subtitle timing math, key-moment scoring). Kept
# unpinned within a broad range so it resolves compatibly with whatever
# version faster-whisper wants at install time.
numpy>=1.24,<3

# --- HTTP --------------------------------------------------------------------
# `requests` is used by the drive downloader and by a couple of GitHub
# helpers on the runner side. The Gemini TTS voiceover script deliberately
# uses only urllib from the stdlib so it never depends on this being
# importable, but keeping requests here matches the rest of the pipeline.
requests>=2.32,<3

# --- Voiceover TTS (Gemini API) ---------------------------------------------
#
# Stage B synthesizes voiceovers with Google's Gemini API TTS. The call is
# a plain HTTPS POST implemented with urllib from the standard library —
# no third-party SDK is required and, deliberately, no Gemini SDK is
# pinned here. This keeps the install lean and lets stage-b.yml skip the
# ~6 GB Chatterbox model cache and the setuptools<81 / pkuseg-from-sdist
# workaround that Chatterbox previously needed.
#
# Gemini API keys are supplied at run time via the GEMINI_API_KEYS env
# variable, populated by stage-b.yml from a GitHub Actions repository
# secret managed by the site's Settings panel (Gemini TTS keys).
# Voice character / model choice / directorial style prompt live in
# scripts/generate_voiceover.py and were selected specifically to
# reproduce a Jujutsu Kaisen / Hunter x Hunter narrator delivery.
